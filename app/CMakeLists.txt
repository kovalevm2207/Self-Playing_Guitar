cmake_minimum_required(VERSION 3.21)
project(SelfGuitar VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GUITAR_CORE_ONLY "Build protocol tests without Qt" OFF)
enable_testing()
add_executable(score_tests tests/score_tests.cpp)
add_test(NAME score_protocol COMMAND score_tests)
add_executable(receiver_tests tests/receiver_tests.cpp)
target_include_directories(receiver_tests PRIVATE tests/stubs)
add_test(NAME firmware_parser COMMAND receiver_tests)
if(GUITAR_CORE_ONLY)
    return()
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 SerialPort)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(SelfGuitar main.cpp backend.cpp backend.h)
qt_add_qml_module(SelfGuitar URI Guitar VERSION 1.0 QML_FILES qml/Main.qml)
qt_add_resources(SelfGuitar catalog PREFIX "/" FILES data/songs.json)
target_link_libraries(SelfGuitar PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::SerialPort)
set_target_properties(SelfGuitar PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
install(TARGETS SelfGuitar BUNDLE DESTINATION . RUNTIME DESTINATION bin)
